```python
import pandas as pd
from rl_trading_lab.domain.ports.market_data import MarketDataPort, Bar, FeatureWindow

class ParquetMarketDataAdapter(MarketDataPort):
    """Adapts pandas DataFrame to MarketDataPort interface."""

    def __init__(self, df: pd.DataFrame):
        self._df = df
        self._validate()

    def _validate(self):
        required = ['open', 'high', 'low', 'close', 'volume']
        missing = [c for c in required if c not in self._df.columns]
        if missing:
            raise ValueError(f"Missing required columns: {missing}")

    def get_price(self, index: int, column: str = "close") -> float:
        return float(self._df.iloc[index][column])

    def get_bar(self, index: int) -> Bar:
        row = self._df.iloc[index]
        return Bar(
            open=float(row['open']),
            high=float(row['high']),
            low=float(row['low']),
            close=float(row['close']),
            volume=float(row['volume']),
            timestamp=row.get('timestamp'),
        )

    def get_feature_window(
        self,
        start_index: int,
        end_index: int,
        features: List[str]
    ) -> FeatureWindow:
        window = self._df.iloc[start_index:end_index][features].values
        return FeatureWindow(
            values=tuple(tuple(row) for row in window),
            feature_names=tuple(features),
        )

    def __len__(self) -> int:
        return len(self._df)
```

**Rationale**: Implements port using pandas, keeps pandas in infrastructure layer.

```python
@dataclass(frozen=True)
class Position:
    """Immutable position value object."""
    size: float = 0.0
    entry_price: float = 0.0
    entry_bar: int = 0

    @classmethod
    def flat(cls) -> 'Position':
        """Factory for flat (no position) state."""
        return cls(size=0.0, entry_price=0.0, entry_bar=0)

    @classmethod
    def open_long(cls, size: float, price: float, bar: int) -> 'Position':
        """Factory for opening long position."""
        return cls(size=abs(size), entry_price=price, entry_bar=bar)

    @classmethod
    def open_short(cls, size: float, price: float, bar: int) -> 'Position':
        """Factory for opening short position."""
        return cls(size=-abs(size), entry_price=price, entry_bar=bar)

    @property
    def is_flat(self) -> bool:
        return self.size == 0.0

    @property
    def is_long(self) -> bool:
        return self.size > 0

    @property
    def is_short(self) -> bool:
        return self.size < 0

    def unrealized_pnl(self, current_price: float) -> float:
        """Calculate unrealized P&L at current price."""
        if self.is_flat:
            return 0.0
        return self.size * (current_price - self.entry_price)
```

**Migration**: Update `Portfolio` to create new `Position` instances instead of mutating.

### 1.4 Enforce Cash Invariants

**File**: `src/rl_trading_lab/environment/portfolio.py` (MODIFY)

**Current** (line 80-92):
```python
def debit(self, amount: float) -> float:
    amount_abs = abs(amount)
    self._balance -= amount_abs  # Can go negative!
    return amount_abs
```

**Proposed**:
```python
class InsufficientFundsError(Exception):
    """Raised when attempting to debit more than available balance."""
    pass

class Cash:
    """Manages cash balance with enforced invariants."""

    def debit(self, amount: float) -> float:
        """
        Remove cash from account.

        Raises:
            InsufficientFundsError: If amount exceeds balance
        """
        amount_abs = abs(amount)
        if amount_abs > self._balance:
            raise InsufficientFundsError(
                f"Cannot debit ${amount_abs:.2f} from balance ${self._balance:.2f}"
            )
        self._balance -= amount_abs
        return amount_abs

    def try_debit(self, amount: float) -> tuple[bool, float]:
        """
        Attempt to debit, returning success status.

        Returns:
            Tuple of (success, amount_debited)
        """
        amount_abs = abs(amount)
        if amount_abs > self._balance:
            return (False, 0.0)
        self._balance -= amount_abs
        return (True, amount_abs)
```

**Rationale**: Domain invariant "cash cannot be negative" must be enforced.

```python
from dataclasses import dataclass
from datetime import datetime
from typing import Optional
from enum import Enum

class TradeSide(Enum):
    LONG = "LONG"
    SHORT = "SHORT"

@dataclass(frozen=True)
class CompletedTrade:
    """Immutable record of a completed round-trip trade."""
    trade_id: int
    side: TradeSide
    entry_price: float
    exit_price: float
    position_size: float
    entry_bar: int
    exit_bar: int
    entry_commission: float
    exit_commission: float
    entry_timestamp: Optional[datetime] = None
    exit_timestamp: Optional[datetime] = None

    @property
    def gross_pnl(self) -> float:
        """P&L before commissions."""
        return self.position_size * (self.exit_price - self.entry_price)

    @property
    def net_pnl(self) -> float:
        """P&L after commissions."""
        return self.gross_pnl - self.entry_commission - self.exit_commission

    @property
    def return_pct(self) -> float:
        """Return as percentage of trade value."""
        entry_value = abs(self.position_size * self.entry_price)
        if entry_value == 0:
            return 0.0
        return self.net_pnl / entry_value

    @property
    def hold_bars(self) -> int:
        """Number of bars position was held."""
        return self.exit_bar - self.entry_bar
```

**Rationale**: Replace dict-based trade history with proper value objects.

### 2.1 Create Pure TradingDomain Class

**File**: `src/rl_trading_lab/domain/trading_domain.py` (NEW)

```python
from dataclasses import dataclass
from typing import Optional, List, Tuple
from enum import IntEnum

from rl_trading_lab.domain.ports.market_data import MarketDataPort
from rl_trading_lab.domain.value_objects.position import Position
from rl_trading_lab.domain.value_objects.trade import CompletedTrade

class OrderIntent(IntEnum):
    """Trading order intents."""
    HOLD = 0
    BUY = 1
    SELL = 2

@dataclass
class TradingState:
    """Current state of the trading domain."""
    current_bar: int
    position: Position
    cash_balance: float
    portfolio_value: float
    is_terminated: bool
    termination_reason: Optional[str] = None

@dataclass
class StepResult:
    """Result of processing an order intent."""
    new_state: TradingState
    reward: float
    trade_executed: Optional[CompletedTrade] = None

class TradingDomain:
    """
    Pure domain logic for trading simulation.

    No external framework dependencies (no gym, no pandas).
    All data access goes through MarketDataPort.
    """

    def __init__(
        self,
        market_data: MarketDataPort,
        initial_balance: float,
        commission_rate: float,
        slippage_rate: float,
        lookback_window: int,
        observation_features: List[str],
        # Domain services (injected)
        position_sizing: 'PositionSizingService',
        reward_calculator: 'RewardCalculationService',
        risk_manager: 'RiskManagementService',
    ):
        self._market_data = market_data
        self._initial_balance = initial_balance
        self._commission_rate = commission_rate
        self._slippage_rate = slippage_rate
        self._lookback_window = lookback_window
        self._observation_features = observation_features

        # Domain services
        self._position_sizing = position_sizing
        self._reward_calculator = reward_calculator
        self._risk_manager = risk_manager

        # State
        self._portfolio: Portfolio = None
        self._current_bar: int = 0
        self._trade_history: List[CompletedTrade] = []

    def reset(self, start_bar: Optional[int] = None) -> TradingState:
        """Reset domain to initial state."""
        self._portfolio = Portfolio(
            initial_cash=self._initial_balance,
            commission_rate=self._commission_rate,
            slippage_rate=self._slippage_rate,
        )
        self._current_bar = start_bar or self._lookback_window
        self._trade_history = []
        return self._get_state()

    def process_order(self, intent: OrderIntent) -> StepResult:
        """Process an order intent and advance one bar."""
        prev_value = self._get_portfolio_value()

        # Execute order through portfolio
        trade = self._execute_intent(intent)
        if trade:
            self._trade_history.append(trade)

        # Advance bar
        self._current_bar += 1

        # Calculate reward
        current_value = self._get_portfolio_value()
        reward = self._reward_calculator.calculate(
            prev_value=prev_value,
            current_value=current_value,
        )

        # Check termination
        is_terminated, reason = self._risk_manager.check_termination(
            portfolio_value=current_value,
            initial_balance=self._initial_balance,
            history=self._trade_history,
        )

        return StepResult(
            new_state=self._get_state(is_terminated, reason),
            reward=reward,
            trade_executed=trade,
        )

    def get_observation(self) -> Tuple[float, ...]:
        """Get current observation as flat tuple."""
        # Feature window
        window = self._market_data.get_feature_window(
            start_index=self._current_bar - self._lookback_window,
            end_index=self._current_bar + 1,
            features=self._observation_features,
        )

        # Position info
        current_price = self._market_data.get_price(self._current_bar)
        position_pnl = self._portfolio.position.unrealized_pnl(current_price)
        portfolio_ratio = self._get_portfolio_value() / self._initial_balance

        return window.flatten() + (
            self._portfolio.position.size,
            position_pnl,
            portfolio_ratio,
        )

    def get_valid_actions(self) -> Tuple[bool, bool, bool]:
        """Get mask of valid actions (HOLD, BUY, SELL)."""
        position = self._portfolio.position
        if position.is_flat:
            return (True, True, True)
        elif position.is_long:
            return (True, False, True)  # Can't buy more
        else:
            return (True, True, False)  # Can't sell more

    # ... private methods
```

**Rationale**: Pure domain class with no framework dependencies.

```python
import numpy as np
import gymnasium as gym
from gymnasium import spaces
from typing import Optional, Dict, Any, Tuple

from rl_trading_lab.domain.trading_domain import TradingDomain, OrderIntent

class GymTradingEnvAdapter(gym.Env):
    """
    Anti-Corruption Layer adapting TradingDomain to Gymnasium interface.

    This adapter:
    - Translates gym.Env interface to domain operations
    - Converts numpy arrays to/from domain types
    - Handles gym-specific metadata and rendering
    """

    metadata = {'render_modes': ['human']}

    def __init__(self, domain: TradingDomain, randomize_start: bool = True):
        super().__init__()
        self._domain = domain
        self._randomize_start = randomize_start

        # Define spaces based on domain configuration
        self.action_space = spaces.Discrete(3)  # HOLD, BUY, SELL

        obs_dim = domain.observation_dimension
        self.observation_space = spaces.Box(
            low=-np.inf, high=np.inf,
            shape=(obs_dim,),
            dtype=np.float32
        )

    def reset(
        self,
        *,
        seed: Optional[int] = None,
        options: Optional[dict] = None
    ) -> Tuple[np.ndarray, Dict[str, Any]]:
        super().reset(seed=seed)

        # Determine start bar
        start_bar = None
        if self._randomize_start:
            max_start = len(self._domain) - self._domain.min_episode_length
            start_bar = self.np_random.integers(
                self._domain.lookback_window,
                max_start
            )

        # Reset domain
        state = self._domain.reset(start_bar=start_bar)

        # Get observation
        obs = np.array(self._domain.get_observation(), dtype=np.float32)
        info = self._state_to_info(state)

        return obs, info

    def step(self, action: int) -> Tuple[np.ndarray, float, bool, bool, Dict[str, Any]]:
        # Convert gym action to domain intent
        intent = OrderIntent(action)

        # Process in domain
        result = self._domain.process_order(intent)

        # Convert back to gym format
        obs = np.array(self._domain.get_observation(), dtype=np.float32)
        reward = float(result.reward)
        terminated = result.new_state.is_terminated
        truncated = self._domain.current_bar >= len(self._domain) - 1
        info = self._state_to_info(result.new_state)

        if result.trade_executed:
            info['trade'] = {
                'side': result.trade_executed.side.value,
                'pnl': result.trade_executed.net_pnl,
            }

        return obs, reward, terminated, truncated, info

    def action_masks(self) -> np.ndarray:
        """For MaskablePPO compatibility."""
        valid = self._domain.get_valid_actions()
        return np.array(valid, dtype=bool)

    def _state_to_info(self, state) -> Dict[str, Any]:
        return {
            'step': state.current_bar,
            'cash': state.cash_balance,
            'portfolio_value': state.portfolio_value,
            'position': state.position.size,
        }

    def render(self):
        # Delegate to domain or implement here
        pass
```

**Rationale**: Clean separation between domain and Gymnasium framework.

```python
from abc import ABC, abstractmethod
from enum import Enum

class RewardType(Enum):
    RETURNS = "returns"
    PNL = "pnl"

class RewardCalculationService(ABC):
    """Domain service for calculating step rewards."""

    @abstractmethod
    def calculate(self, prev_value: float, current_value: float) -> float:
        ...

class ReturnsRewardCalculation(RewardCalculationService):
    """Reward based on percentage returns."""

    def __init__(self, clip_min: float = -10.0, clip_max: float = 10.0):
        self._clip_min = clip_min
        self._clip_max = clip_max

    def calculate(self, prev_value: float, current_value: float) -> float:
        if prev_value == 0:
            return 0.0
        reward = (current_value - prev_value) / prev_value
        return max(self._clip_min, min(self._clip_max, reward))

class PnLRewardCalculation(RewardCalculationService):
    """Reward based on absolute P&L."""

    def __init__(self, clip_min: float = -10.0, clip_max: float = 10.0):
        self._clip_min = clip_min
        self._clip_max = clip_max

    def calculate(self, prev_value: float, current_value: float) -> float:
        reward = current_value - prev_value
        return max(self._clip_min, min(self._clip_max, reward))
```

```python
from abc import ABC, abstractmethod
from typing import Tuple, Optional, List
from dataclasses import dataclass

from rl_trading_lab.domain.value_objects.trade import CompletedTrade

@dataclass(frozen=True)
class RiskLimits:
    """Configuration for risk management."""
    max_drawdown_pct: float = 0.30  # 30%
    min_portfolio_pct: float = 0.20  # 20% of initial
    max_consecutive_losses: Optional[int] = None

class RiskManagementService(ABC):
    """Domain service for risk checks and termination."""

    @abstractmethod
    def check_termination(
        self,
        portfolio_value: float,
        initial_balance: float,
        history: List[CompletedTrade],
        peak_value: Optional[float] = None,
    ) -> Tuple[bool, Optional[str]]:
        """
        Check if episode should terminate due to risk limits.

        Returns:
            Tuple of (should_terminate, reason)
        """
        ...

class StandardRiskManagement(RiskManagementService):
    """Standard risk management with configurable limits."""

    def __init__(self, limits: RiskLimits):
        self._limits = limits
        self._peak_value: float = 0.0

    def check_termination(
        self,
        portfolio_value: float,
        initial_balance: float,
        history: List[CompletedTrade],
        peak_value: Optional[float] = None,
    ) -> Tuple[bool, Optional[str]]:
        # Update peak
        if peak_value:
            self._peak_value = max(self._peak_value, peak_value)
        self._peak_value = max(self._peak_value, portfolio_value)

        # Check minimum portfolio value
        min_value = initial_balance * self._limits.min_portfolio_pct
        if portfolio_value < min_value:
            return (True, f"Portfolio below {self._limits.min_portfolio_pct:.0%} of initial")

        # Check drawdown
        if self._peak_value > 0:
            drawdown = (self._peak_value - portfolio_value) / self._peak_value
            if drawdown > self._limits.max_drawdown_pct:
                return (True, f"Drawdown exceeded {self._limits.max_drawdown_pct:.0%}")

        # Check consecutive losses
        if self._limits.max_consecutive_losses:
            consecutive = self._count_consecutive_losses(history)
            if consecutive >= self._limits.max_consecutive_losses:
                return (True, f"Exceeded {self._limits.max_consecutive_losses} consecutive losses")

        return (False, None)

    def _count_consecutive_losses(self, history: List[CompletedTrade]) -> int:
        count = 0
        for trade in reversed(history):
            if trade.net_pnl < 0:
                count += 1
            else:
                break
        return count

    def reset(self):
        """Reset peak tracking for new episode."""
        self._peak_value = 0.0
```

```python
        # 1. Create environments
        train_env = self._env_service.create_training_env(config)
        eval_env = self._env_service.create_eval_env(config)

        # 2. Create agent
        agent = self._agent_service.create_agent(
            config=config.agent,
            env=train_env,
            device=config.experiment.device,
        )

        # 3. Setup tracking
        if self._tracker:
            self._tracker.start_run(
                run_name=config.experiment.run_name,
                params=self._extract_params(config),
            )

        # 4. Train
        callbacks = self._checkpoint_service.create_callbacks(
            save_path=config.training.save_path,
            eval_env=eval_env,
            eval_freq=config.training.eval_freq,
        )

        agent.learn(
            total_timesteps=config.training.total_timesteps,
            callback=callbacks,
            progress_bar=config.logging.console.progress_bar,
        )

        # 5. Save final model
        final_path = self._checkpoint_service.save_final(
            agent=agent,
            path=config.training.save_path,
        )

        # 6. End tracking
        if self._tracker:
            self._tracker.end_run()

        return TrainingResult(
            final_model_path=final_path,
            total_timesteps=config.training.total_timesteps,
            best_model_path=self._checkpoint_service.best_model_path,
        )

    def _extract_params(self, config: RootConfig) -> Dict[str, Any]:
        return {
            "agent": config.agent.name,
            "reward_type": config.env.environment_params.reward_type,
            "total_timesteps": config.training.total_timesteps,
        }
```

```python
from dataclasses import dataclass
from typing import Dict, Any, List
from pathlib import Path
import numpy as np

@dataclass
class EvaluationResult:
    """Result of evaluation use case."""
    mean_reward: float
    std_reward: float
    mean_episode_length: float
    sharpe_ratio: float
    max_drawdown: float
    num_trades: int
    win_rate: float
    episode_rewards: List[float]

class EvaluateAgentUseCase:
    """
    Use case: Evaluate a trained agent.

    Runs agent on test environment and computes performance metrics.
    """

    def __init__(
        self,
        environment_service: 'EnvironmentService',
        agent_service: 'AgentService',
    ):
        self._env_service = environment_service
        self._agent_service = agent_service

    def execute(
        self,
        model_path: Path,
        config: 'RootConfig',
        n_episodes: int = 10,
        deterministic: bool = True,
    ) -> EvaluationResult:
        """Execute the evaluation use case."""
        # 1. Create test environment
        test_env = self._env_service.create_test_env(config)

        # 2. Load agent
        agent = self._agent_service.load_agent(model_path, test_env)

        # 3. Run episodes
        episode_rewards = []
        episode_lengths = []
        all_trades = []

        for _ in range(n_episodes):
            obs, info = test_env.reset()
            done = False
            episode_reward = 0.0
            episode_length = 0

            while not done:
                action, _ = agent.predict(obs, deterministic=deterministic)
                obs, reward, terminated, truncated, info = test_env.step(action)
                episode_reward += reward
                episode_length += 1
                done = terminated or truncated

            episode_rewards.append(episode_reward)
            episode_lengths.append(episode_length)

            # Collect trades from environment
            if hasattr(test_env, 'get_trade_history'):
                all_trades.extend(test_env.get_trade_history())

        # 4. Compute metrics
        returns = np.array(episode_rewards)
        sharpe = self._compute_sharpe(returns)
        max_dd = self._compute_max_drawdown(returns)
        win_rate = self._compute_win_rate(all_trades)

        return EvaluationResult(
            mean_reward=float(np.mean(returns)),
            std_reward=float(np.std(returns)),
            mean_episode_length=float(np.mean(episode_lengths)),
            sharpe_ratio=sharpe,
            max_drawdown=max_dd,
            num_trades=len(all_trades),
            win_rate=win_rate,
            episode_rewards=episode_rewards,
        )

    def _compute_sharpe(self, returns: np.ndarray, annualization: int = 252) -> float:
        if len(returns) == 0 or returns.std() < 1e-8:
            return 0.0
        return returns.mean() / returns.std() * np.sqrt(annualization)

    def _compute_max_drawdown(self, returns: np.ndarray) -> float:
        cumulative = np.cumsum(returns)
        peak = np.maximum.accumulate(cumulative)
        drawdown = (peak - cumulative) / (peak + 1e-8)
        return float(drawdown.max())

    def _compute_win_rate(self, trades: List) -> float:
        if not trades:
            return 0.0
        wins = sum(1 for t in trades if t.get('net_pnl', 0) > 0)
        return wins / len(trades)
```
